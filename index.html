function doGet() { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = sheet.getDataRange().getValues(); const result = []; for (let i = 1; i < data.length; i++) { const lastName = data[i][0] || ""; const firstName = data[i][1] || ""; const fullName = (lastName + " " + firstName).trim(); if (!fullName) continue; const salon = data[i][2] || ""; const status = data[i][3] || "未"; // E列の更新日時を取得（日付オブジェクトの場合はフォーマット変換） let updatedAt = data[i][4] || ""; if (updatedAt instanceof Date) { updatedAt = Utilities.formatDate(updatedAt, "JST", "yyyy/MM/dd HH:mm"); } result.push({ id: i + 1, name: fullName, salon: salon, status: status, updatedAt: updatedAt }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const params = JSON.parse(e.postData.contents); if (params.row && params.status) { const now = Utilities.formatDate(new Date(), "JST", "yyyy/MM/dd HH:mm"); // D列（参加状況）と E列（更新日時）をセットで更新 sheet.getRange(params.row, 4).setValue(params.status); sheet.getRange(params.row, 5).setValue(now); } return ContentService.createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } // スプレッドシートを直接手入力で書き換えた際にもE列に日時を自動セットするトリガー function onEdit(e) { const range = e.range; const sheet = range.getSheet(); // D列（4列目）が変更された場合 if (range.getColumn() === 4 && range.getRow() > 1) { const now = Utilities.formatDate(new Date(), "JST", "yyyy/MM/dd HH:mm"); sheet.getRange(range.getRow(), 5).setValue(now); } }
